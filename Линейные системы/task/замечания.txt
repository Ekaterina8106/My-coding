void init_matrix(int n, double** A, int s, char* filename) {
    if (s != 0) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                A[i][j] = f(s, n, i, j);
            }
        }
    }
    else {
        FILE* file = fopen(filename, "r");
        if (!file) {
            printf("Ошибка открытия файла %s\n", filename);
            exit(1);
        }

        // проверка на корректность ввода матрицы
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (fscanf(file, "%lf", &A[i][j]) != 1) {
                    printf("Ошибка: в файле недостаточно данных или неверный формат\n");
                    fclose(file); // ЗАКРЫВАЕМ файл перед выходом!
                    exit(1);
                }
            }
        }

        // ПРОВЕРКА НА ЛИШНИЕ ДАННЫЕ В ФАЙЛЕ
        double extra_value;
        if (fscanf(file, "%lf", &extra_value) == 1) {
            printf("Предупреждение: в файле больше данных, чем требуется для матрицы %dx%d\n", n, n);
            // Можно считать ошибкой или предупреждением - по условию задачи
            printf("Ошибка: в файле больше элементов, чем требуется\n");
            fclose(file);
            exit(1);
        }

        fclose(file);

}
